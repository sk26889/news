// ==================== VALIDASI FORMAT WAKTU ====================
function validateDateTime($dateTime) {
    // Format: YYYY-MM-DD HH:MM:SS
    $pattern = '/^\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2}$/';
    if (!preg_match($pattern, $dateTime)) {
        return false;
    }
    
    $parts = explode(' ', $dateTime);
    $date = explode('-', $parts[0]);
    $time = explode(':', $parts[1]);
    
    $year = (int)$date[0];
    $month = (int)$date[1];
    $day = (int)$date[2];
    $hour = (int)$time[0];
    $minute = (int)$time[1];
    $second = (int)$time[2];
    
    // Validasi range
    if ($year < 1970 || $year > 2038) return false;
    if ($month < 1 || $month > 12) return false;
    if ($day < 1 || $day > 31) return false;
    if ($hour < 0 || $hour > 23) return false;
    if ($minute < 0 || $minute > 59) return false;
    if ($second < 0 || $second > 59) return false;
    
    // Validasi tanggal (termasuk leap year)
    return checkdate($month, $day, $year);
}

// ==================== KONVERSI KE TIMESTAMP ====================
function convertToTimestamp($dateTime) {
    return strtotime($dateTime);
}

// ==================== FUNGSI UTAMA MASS CHANGE ====================
function massChangeTimestamp($directory, $customDateTime, $recursive = true, $showProgress = true) {
    
    if (!is_dir($directory)) {
        return ["error" => "Directory '$directory' tidak ditemukan!"];
    }
    
    $timestamp = convertToTimestamp($customDateTime);
    if ($timestamp === false) {
        return ["error" => "Format tanggal/waktu '$customDateTime' tidak valid!"];
    }
    
    $result = [
        'total_files' => 0,
        'total_dirs' => 0,
        'success_files' => 0,
        'success_dirs' => 0,
        'failed' => [],
        'target_date' => date('Y-m-d H:i:s', $timestamp),
        'timestamp' => $timestamp
    ];
    
    echo YELLOW . "\n[!] MEMULAI PROSES MASS TIMESTAMP" . RESET . "\n";
    echo "Target Directory: " . realpath($directory) . "\n";
    echo "Target DateTime: " . CYAN . date('Y-m-d H:i:s', $timestamp) . RESET . "\n";
    echo "Format: Tahun-Bulan-Tanggal Jam:Menit:Detik (24 jam)\n";
    echo str_repeat("-", 60) . "\n\n";
    
    try {
        if ($recursive) {
            $iterator = new RecursiveIteratorIterator(
                new RecursiveDirectoryIterator($directory, RecursiveDirectoryIterator::SKIP_DOTS),
                RecursiveIteratorIterator::SELF_FIRST
            );
        } else {
            $iterator = new FilesystemIterator($directory);
        }
        
        $count = 0;
        foreach ($iterator as $item) {
            $path = $item->getPathname();
            $count++;
            
            if ($item->isDir()) {
                $result['total_dirs']++;
                if (@touch($path, $timestamp, $timestamp)) {
                    $result['success_dirs']++;
                } else {
                    $result['failed'][] = "[DIR] " . $path;
                }
            } else {
                $result['total_files']++;
                if (@touch($path, $timestamp, $timestamp)) {
                    $result['success_files']++;
                } else {
                    $result['failed'][] = "[FILE] " . $path;
                }
            }
            
            if ($showProgress && $count % 100 == 0) {
                echo "  Progress: " . CYAN . $count . RESET . " item diproses...\n";
            }
        }
        
    } catch (Exception $e) {
        $result['error'] = $e->getMessage();
    }
    
    return $result;
}

// ==================== TAMPILKAN HASIL ====================
function showResult($result) {
    if (isset($result['error'])) {
        echo RED . "\n✗ ERROR: " . $result['error'] . "\n" . RESET;
        return;
    }
    
    echo "\n" . str_repeat("=", 60) . "\n";
    echo GREEN . "✓ PROSES SELESAI!" . RESET . "\n";
    echo str_repeat("-", 60) . "\n";
    echo "Target Date & Time: " . CYAN . $result['target_date'] . RESET . "\n";
    echo "Format: " . YELLOW . "YYYY-MM-DD HH:MM:SS (24 jam dengan detik)" . RESET . "\n";
    echo str_repeat("-", 60) . "\n";
    echo "Total Folder : " . $result['total_dirs'] . " (berhasil: " . GREEN . $result['success_dirs'] . RESET . ")\n";
    echo "Total File   : " . $result['total_files'] . " (berhasil: " . GREEN . $result['success_files'] . RESET . ")\n";
    echo "Total Item   : " . ($result['total_files'] + $result['total_dirs']) . "\n";
    
    if (!empty($result['failed'])) {
        echo RED . "\n⚠️ GAGAL (" . count($result['failed']) . " item):\n" . RESET;
        foreach (array_slice($result['failed'], 0, 20) as $fail) {
            echo "  - $fail\n";
        }
        if (count($result['failed']) > 20) {
            echo YELLOW . "  ... dan " . (count($result['failed']) - 20) . " error lainnya\n" . RESET;
        }
    }
    
    echo "\n" . GREEN . "✓ Semua timestamp berhasil diubah ke: " . $result['target_date'] . RESET . "\n";
}

    echo "  ✅ 2024-12-31 23:59:59 (Tahun baru 2025)\n";
    echo "  ✅ 2020-01-01 00:00:00 (Awal tahun 2020)\n";
    echo "  ✅ 2023-06-15 14:30:45 (Tengah tahun, jam 2 siang)\n";
    echo "  ✅ 2021-12-25 08:15:30 (Hari Natal pagi)\n\n";
    
    echo str_repeat("=", 60) . "\n";
    
    // Input target directory
    echo CYAN . "\n📁 Masukkan target directory: " . RESET;
    $targetDir = trim(fgets(STDIN));
    
    if (empty($targetDir)) {
        die(RED . "Error: Directory tidak boleh kosong!\n" . RESET);
    }
    
    if (!is_dir($targetDir)) {
        die(RED . "Error: Directory '$targetDir' tidak ditemukan!\n" . RESET);
    }
    
    // Input custom date time (dengan detik)
    echo CYAN . "📅 Masukkan tanggal & waktu (YYYY-MM-DD HH:MM:SS): " . RESET;
    $customDateTime = trim(fgets(STDIN));
    
    if (empty($customDateTime)) {
        die(RED . "Error: Tanggal & waktu tidak boleh kosong!\n" . RESET);
    }
    
    // Validasi format
    if (!validateDateTime($customDateTime)) {
        die(RED . "Error: Format '$customDateTime' tidak valid!\n" . RESET .
            "Gunakan format: YYYY-MM-DD HH:MM:SS (contoh: 2024-12-31 23:59:59)\n");
    }
    
    $timestamp = convertToTimestamp($customDateTime);
    $formattedDate = date('Y-m-d H:i:s', $timestamp);
    
    // Tampilkan informasi
    echo "\n" . str_repeat("-", 60) . "\n";
    echo YELLOW . "⚠️  KONFIRMASI PROSES" . RESET . "\n";
    echo "Target Directory: " . realpath($targetDir) . "\n";

    echo "Detik           : " . date('s', $timestamp) . "\n";

        die(YELLOW . "Dibatalkan.\n" . RESET);

    $result = massChangeTimestamp($targetDir, $customDateTime, true, true);
    showResult($result);
